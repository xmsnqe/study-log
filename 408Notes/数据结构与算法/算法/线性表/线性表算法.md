## 📝算法详解

### 一、线性表类型定义

```c
#define OK 1
#define ERROR 0
#define TRUE 1
#define FALSE 0
#define OVERFLOW -2
#define INIT_SIZE 100    // 初始分配存储空间容量
#define INCREM 10         // 存储空间分配增量

typedef int Status;
typedef int ElemType;

/* ---------- 顺序表定义 ---------- */
typedef struct {
    ElemType *elem;   // 存储空间基址
    int length;       // 当前长度
    int listsize;     // 当前分配的存储容量（以sizeof(ElemType)为单位）
} SqList;

/* ---------- 单链表结点定义 ---------- */
typedef struct LNode {
    ElemType data;
    struct LNode *next;
} LNode, *LinkList;

/* ---------- 双向链表结点定义 ---------- */
typedef struct DLNode {
    ElemType data;
    struct DLNode *prior, *next;
} DLNode, *DuLinkList;

/* ---------- 静态链表结点定义 ---------- */
#define MAXSIZE 1000
typedef struct {
    ElemType data;
    int cur;   // 游标，指示后继元素在数组中的位置
} Component, StaticList[MAXSIZE];
```

### 二、顺序表基本运算

#### 2.1 顺序表初始化 `InitList_Sq`

```c
/* 操作结果：构造一个空的顺序表L */
Status InitList_Sq(SqList &L) {
    L.elem = (ElemType *)malloc(INIT_SIZE * sizeof(ElemType));
    if (!L.elem) exit(OVERFLOW);   // 存储分配失败
    L.length = 0;                   // 空表长度为0
    L.listsize = INIT_SIZE;         // 初始存储容量
    return OK;
}
```

- **时间复杂度**：O(1)
- **空间复杂度**：O(INIT_SIZE)，分配初始空间

#### 2.2 顺序表销毁 `DestroyList_Sq`

```c
/* 初始条件：顺序表L已存在。操作结果：销毁顺序表L */
Status DestroyList_Sq(SqList &L) {
    if (L.elem) {
        free(L.elem);
        L.elem = NULL;
        L.length = 0;
        L.listsize = 0;
    }
    return OK;
}
```

#### 2.3 按位查找 `GetElem_Sq`

```c
/* 初始条件：顺序表L已存在，1≤i≤ListLength_Sq(L) */
/* 操作结果：用e返回L中第i个数据元素的值 */
Status GetElem_Sq(SqList L, int i, ElemType &e) {
    if (i < 1 || i > L.length) return ERROR;  // i值不合法
    e = L.elem[i - 1];   // 第i个元素的存储地址为 elem[i-1]
    return OK;
}
```

- **时间复杂度**：⭐ O(1) —— 随机存取
- **空间复杂度**：O(1)

#### 2.4 按值查找 `LocateElem_Sq`

```c
/* 初始条件：顺序表L已存在，compare()为数据元素判定函数 */
/* 操作结果：返回L中第一个与e满足compare()关系的元素的位序 */
/*           若这样的元素不存在，则返回值为0 */
int LocateElem_Sq(SqList L, ElemType e, Status (*compare)(ElemType, ElemType)) {
    int i = 1;       // i的初值为第一个元素的位序
    ElemType *p = L.elem;  // p的初值为第一个元素的存储位置
    
    while (i <= L.length && !(*compare)(*p++, e))
        ++i;
    
    if (i <= L.length)
        return i;      // 找到，返回位序
    else
        return 0;      // 未找到
}
```

**等价简化版（ElemType为int，直接比较）**：

```c
int LocateElem_Sq_Int(SqList L, ElemType e) {
    for (int i = 0; i < L.length; i++) {
        if (L.elem[i] == e)
            return i + 1;  // 返回位序（从1开始）
    }
    return 0;  // 未找到
}
```

- **时间复杂度**：⭐ O(n) —— 最坏查找所有元素
- **空间复杂度**：O(1)

#### 2.5 顺序表插入 `ListInsert_Sq` 🔥⭐

```c
/* 初始条件：顺序表L已存在，1≤i≤ListLength_Sq(L)+1 */
/* 操作结果：在L中第i个位置之前插入新的数据元素e，L的长度+1 */
Status ListInsert_Sq(SqList &L, int i, ElemType e) {
    if (i < 1 || i > L.length + 1) return ERROR;  // i值不合法
    
    if (L.length >= L.listsize) {  // 当前存储空间已满，增加分配
        ElemType *newbase = (ElemType *)realloc(L.elem,
                    (L.listsize + INCREM) * sizeof(ElemType));
        if (!newbase) exit(OVERFLOW);
        L.elem = newbase;                // 新基址
        L.listsize += INCREM;            // 增加存储容量
    }
    
    ElemType *q = &(L.elem[i - 1]);  // q为插入位置
    ElemType *p;
    
    // 插入位置及之后的元素右移
    for (p = &(L.elem[L.length - 1]); p >= q; --p) {
        *(p + 1) = *p;
    }
    
    *q = e;       // 插入e
    ++L.length;     // 表长+1
    return OK;
}
```

**插入过程图解**：

```
插入前：  [a1][a2][a3]    [a4]    [a5]
          ^  ^  ^
          1  2  3   4     5
          
在第3个位置插入e：

插入过程：
  Step1: 将a5右移到位置5
  Step2: 将a4右移到位置4
  Step3: 将e插入位置3

插入后：  [a1][a2][a3][e ]    [a4]    [a5]
          1    2    3  4      5       6
```

- **时间复杂度**：🔥⭐
  - 最好情况（i = n+1，表尾插入）：O(1)
  - 最坏情况（i = 1，表头插入）：O(n)
  - 平均情况：移动次数期望值 = n/2，故 **O(n)**
- **空间复杂度**：O(1)（不含存储空间扩容）
- **考点**：408常考平均移动次数计算

#### 2.6 顺序表删除 `ListDelete_Sq` 🔥⭐

```c
/* 初始条件：顺序表L已存在且非空，1≤i≤ListLength_Sq(L) */
/* 操作结果：删除L的第i个数据元素，并用e返回其值，L的长度-1 */
Status ListDelete_Sq(SqList &L, int i, ElemType &e) {
    if (i < 1 || i > L.length) return ERROR;  // i值不合法
    
    ElemType *p = &(L.elem[i - 1]);   // p为被删除元素的位置
    e = *p;                            // 被删除元素的值赋给e
    
    ElemType *q = L.elem + L.length - 1;  // q为表尾元素的位置
    
    // 被删除元素之后的元素左移
    for (++p; p <= q; ++p) {
        *(p - 1) = *p;
    }
    
    --L.length;     // 表长-1
    return OK;
}
```

**删除过程图解**：

```
删除前：  [a1][a2][a3]    [a4]    [a5]
          1    2    3  4      5
          
删除第3个位置元素：

删除过程：
  Step1: 将a4左移到位置3
  Step2: 将a5左移到位置4

删除后：  [a1][a2][a4]    [a5]    (a3)
          1    2    3      4       被删除
```

- **时间复杂度**：🔥⭐
  - 最好情况（i = n，删除表尾）：O(1)
  - 最坏情况（i = 1，删除表头）：O(n)
  - 平均情况：移动次数期望值 = (n-1)/2，故 **O(n)**
- **空间复杂度**：O(1)

### 三、单链表基本运算

#### 3.1 头插法建立单链表 `CreateList_F`

```c
/* 逆位序输入n个元素的值，建立带表头结点的单链表L */
void CreateList_F(LinkList &L, int n) {
    L = (LinkList)malloc(sizeof(LNode));  // 建立头结点
    L->next = NULL;                       // 先建立一个带头结点的空链表
    
    for (int i = 0; i < n; ++i) {
        LinkList p = (LinkList)malloc(sizeof(LNode));  // 生成新结点
        printf("请输入第%d个元素: ", i + 1);
        scanf("%d", &p->data);
        
        p->next = L->next;  // 将新结点插入表头
        L->next = p;
    }
}
```

**头插法特点**：
- 新结点始终插入在头结点之后
- 链表中元素顺序与输入顺序**相反**
- 时间复杂度：**O(n)**

---

#### 3.2 尾插法建立单链表 `CreateList_R`

```c
/* 正位序输入n个元素的值，建立带表头结点的单链表L */
void CreateList_R(LinkList &L, int n) {
    L = (LinkList)malloc(sizeof(LNode));  // 建立头结点
    L->next = NULL;
    
    LinkList r = L;  // r为表尾指针，指向当前链表最后一个结点
    
    for (int i = 0; i < n; ++i) {
        LinkList p = (LinkList)malloc(sizeof(LNode));  // 生成新结点
        printf("请输入第%d个元素: ", i + 1);
        scanf("%d", &p->data);
        
        p->next = NULL;
        r->next = p;   // 将新结点插入r之后
        r = p;         // r指向新的表尾结点
    }
}
```

**尾插法特点**：
- 新结点始终插入在表尾
- 链表中元素顺序与输入顺序**相同**
- 需要维护尾指针 r
- 时间复杂度：**O(n)**

#### 3.3 单链表按号查找 `GetElem_L`

```c
/* L为带头结点的单链表，查找第i个元素，用e返回其值 */
Status GetElem_L(LinkList L, int i, ElemType &e) {
    LinkList p = L->next;   // p指向第一个结点
    int j = 1;              // j为计数器
    
    while (p && j < i) {    // 顺链向后扫描，直到p为空或找到第i个结点
        p = p->next;
        ++j;
    }
    
    if (!p || j > i) return ERROR;  // i>n或i<1
    e = p->data;
    return OK;
}
```

- **时间复杂度**：O(n)
- **空间复杂度**：O(1)

#### 3.4 单链表按值查找 `LocateElem_L`

```c
/* L为带头结点的单链表，返回第一个与e满足compare()关系的元素的位序 */
int LocateElem_L(LinkList L, ElemType e, Status (*compare)(ElemType, ElemType)) {
    LinkList p = L->next;
    int j = 1;
    
    while (p && !(*compare)(p->data, e)) {
        p = p->next;
        ++j;
    }
    
    if (p) return j;
    else return 0;
}

/* 简化版（直接比较）*/
int LocateElem_L_Int(LinkList L, ElemType e) {
    LinkList p = L->next;
    int j = 1;
    while (p && p->data != e) {
        p = p->next;
        ++j;
    }
    if (p) return j;
    return 0;
}
```

#### 3.5 单链表插入 `ListInsert_L` 🔥⭐

```c
/* 在带头结点的单链表L中第i个位置之前插入元素e */
Status ListInsert_L(LinkList &L, int i, ElemType e) {
    LinkList p = L;
    int j = 0;
    
    // 寻找第i-1个结点（插入位置的前驱）
    while (p && j < i - 1) {
        p = p->next;
        ++j;
    }
    
    if (!p || j > i - 1) return ERROR;  // i>n+1或i<1
    
    LinkList s = (LinkList)malloc(sizeof(LNode));  // 生成新结点
    s->data = e;
    
    // 插入操作：s->next = p->next; p->next = s;
    s->next = p->next;   // 先将新结点的指针域指向被插结点
    p->next = s;          // 再将前驱结点的指针域指向新结点
    // ⚠️注意顺序不能颠倒！
    
    return OK;
}
```

**插入过程图解**：

```
插入前：  ... [p] ----> [a_i] ----> ...
               ^           ^
               |           |
            前驱p       被插位置
            
插入后：  ... [p] ----> [s(新)] ----> [a_i] ----> ...
                     ^              ^
                     |______________|
```

- **时间复杂度**：⭐
  - 查找前驱结点：O(n)
  - 实际插入操作：O(1)
  - 整体：**O(n)**
- **空间复杂度**：O(1)

#### 3.6 单链表删除 `ListDelete_L` 🔥⭐

```c
/* 删除带头结点的单链表L中第i个元素，用e返回其值 */
Status ListDelete_L(LinkList &L, int i, ElemType &e) {
    LinkList p = L;
    int j = 0;
    
    // 寻找第i-1个结点（被删结点的前驱）
    while (p->next && j < i - 1) {
        p = p->next;
        ++j;
    }
    
    if (!(p->next) || j > i - 1) return ERROR;  // 第i个元素不存在
    
    LinkList q = p->next;   // q指向被删除结点
    e = q->data;
    
    p->next = q->next;      // 将q的直接后继挂到p之后
    free(q);                // 释放q结点
    
    return OK;
}
```

**删除过程图解**：

```
删除前：  ... [p] ----> [q(被删)] ----> [a_{i+1}] ----> ...

删除操作：  p->next = q->next;  （将q的后继直接接到p上）

删除后：  ... [p] --------------> [a_{i+1}] ----> ...
                     （q结点被释放）
```

- **时间复杂度**：⭐
  - 查找前驱结点：O(n)
  - 实际删除操作：O(1)
  - 整体：**O(n)**
- **空间复杂度**：O(1)

### 四、循环链表

循环链表的本质是将单链表的尾结点的 `next` 指向头结点，形成环。

**循环链表判断链表为空**：`L->next == L`（有头结点）或 `L == L->next`

**循环链表插入操作**（与单链表类似，但查找头结点用 `p != L` 判断）：

```c
/* 在带头结点的循环单链表中第i个位置之前插入e */
Status ListInsert_Circular(LinkList &L, int i, ElemType e) {
    LinkList p = L;
    int j = 0;
    
    // 循环链表的终止条件为 p->next != L
    while (p->next != L && j < i - 1) {
        p = p->next;
        ++j;
    }
    
    if (j > i - 1) return ERROR;
    
    LinkList s = (LinkList)malloc(sizeof(LNode));
    s->data = e;
    s->next = p->next;
    p->next = s;
    return OK;
}
```

### 五、双向链表基本运算

#### 5.1 双向链表插入 `InsertDulNode` 🔥⭐

```c
/* 在双向链表结点p之后插入新结点s */
Status InsertDulNode(DuLinkList p, DuLinkList s) {
    if (!p || !s) return ERROR;
    
    /* 四步操作（注意顺序）*/
    s->prior = p;               // ① 新结点的前驱指向p
    s->next = p->next;           // ② 新结点的后继指向p的后继
    if (p->next)                 // ③ p后继的前驱指向s（防止插入在尾部）
        p->next->prior = s;
    p->next = s;                 // ④ p的后继指向s
    
    return OK;
}

/* 在位置i插入 */
Status ListInsert_Dul(DuLinkList &L, int i, ElemType e) {
    DuLinkList p = L;
    int j = 0;
    
    while (p && j < i - 1) {
        p = p->next;
        ++j;
    }
    
    if (!p || j > i - 1) return ERROR;
    
    DuLinkList s = (DuLinkList)malloc(sizeof(DLNode));
    s->data = e;
    return InsertDulNode(p, s);
}
```

**插入过程图解**：

```
插入前：  ... <---> [p] <---> [a] <---> ...
                  prior next         prior next

插入s在p之后：
  Step 1: s->prior = p          s的前驱指向p
  Step 2: s->next = p->next     s的后继指向a
  Step 3: p->next->prior = s    a的前驱指向s
  Step 4: p->next = s           p的后继指向s

插入后：  ... <---> [p] <---> [s] <---> [a] <---> ...
                  prior next   prior next   prior next
```

- **时间复杂度**：O(n)（查找位置）+ O(1)（实际插入）= **O(n)**
- **空间复杂度**：O(1)

---

#### 5.2 双向链表删除 `DeleteDulNode` 🔥⭐

```c
/* 删除双向链表中的结点p */
Status DeleteDulNode(DuLinkList p) {
    if (!p) return ERROR;
    
    /* 四步操作（注意顺序）*/
    if (p->prior)              // p不是头结点
        p->prior->next = p->next;  // ① p前驱的后继指向p的后继
    if (p->next)               // p不是尾结点
        p->next->prior = p->prior; // ② p后继的前驱指向p的前驱
    
    free(p);                   // ③ 释放p结点
    return OK;
}

/* 删除位置i的元素 */
Status ListDelete_Dul(DuLinkList L, int i, ElemType &e) {
    DuLinkList p = L->next;  // 跳过头结点
    int j = 1;
    
    while (p && j < i) {
        p = p->next;
        ++j;
    }
    
    if (!p || j > i) return ERROR;
    
    e = p->data;
    return DeleteDulNode(p);
}
```

**删除过程图解**：

```
删除前：  ... <---> [p_prev] <---> [p(被删)] <---> [p_next] <---> ...

删除p：
  Step 1: p->prior->next = p->next    （p_prev的next跳过硬p指向p_next）
  Step 2: p->next->prior = p->prior   （p_next的prior跳过硬p指向p_prev）
  Step 3: free(p)                     （释放p结点）

删除后：  ... <---> [p_prev] --------------> [p_next] <---> ...
```

- **时间复杂度**：O(n)（查找位置）+ O(1)（实际删除）= **O(n)**
- **空间复杂度**：O(1)

### 六、静态链表

```c
/* 静态链表初始化 */
Status InitList_Static(StaticList space) {
    // 将数组所有分量链成一个备用链表
    for (int i = 0; i < MAXSIZE - 1; i++)
        space[i].cur = i + 1;
    space[MAXSIZE - 1].cur = 0;  // 链表结束标记
    return OK;
}

/* 静态链表分配结点（从备用链表头部取）*/
int Malloc_SLL(StaticList space) {
    int i = space[0].cur;  // 备用链表的第一个结点是待分配的
    if (i)
        space[0].cur = space[i].cur;  // 将备用链表头结点的下一个结点作为新的头
    return i;
}

/* 静态链表回收结点（释放到备用链表头部）*/
void Free_SLL(StaticList space, int k) {
    space[k].cur = space[0].cur;   // 将待释放结点插入备用链表头部
    space[0].cur = k;
}
```

## ⚠️算法复杂度分析

| 算法 | 最好 | 最坏 | 平均 | 空间复杂度 | 是否稳定 |
|------|------|------|------|-----------|---------|
| 顺序表按位查找 GetElem_Sq | O(1) | O(1) | O(1) | O(1) | - |
| 顺序表按值查找 LocateElem_Sq | O(1) | O(n) | O(n) | O(1) | - |
| 顺序表插入 ListInsert_Sq | O(1) | O(n) | O(n) | O(1) | 稳定 |

## ⚖️易混淆对比

### 顺序表 vs 单链表

| 对比维度 | 顺序表 (SqList) | 单链表 (LinkList) |
|---------|-----------------|-------------------|
| 存储方式 | 地址连续 | 地址可连续可不连续 |
| 存取方式 | 随机存取 O(1) | 顺序存取 O(n) |
| 插入删除 | 需移动元素 O(n) | 只需修改指针 O(1)（已知位置） |
| 空间分配 | 静态分配/动态扩容 | 动态分配 |
| 存储密度 | 1（无额外指针域） | < 1（每个结点多一个指针域） |
| 适用场景 | 频繁查找，较少插删 | 频繁插删，较少查找 |
| 空间利用率 | 可能浪费（预分配） | 按需分配，但指针开销大 |

### 头插法 vs 尾插法

| 对比维度 | 头插法 | 尾插法 |
|---------|--------|--------|
| 新结点插入位置 | 头结点之后 | 表尾 |
| 链表元素顺序 | 与输入顺序**相反** | 与输入顺序**相同** |
| 是否需要尾指针 | 不需要 | 需要（r指针） |
